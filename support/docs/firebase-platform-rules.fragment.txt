Inside match /databases/{database}/documents:

function platformInstructor() {
  return request.auth != null
    && request.auth.token.email == 'luisceballos@gmail.com'
    && request.auth.token.email_verified == true;
}
function platformPolicy(lectureId) {
  return get(/databases/$(database)/documents/lectures_hub/config)
    .data.accessPolicies[lectureId];
}
function platformLive(lectureId, section) {
  let policy = platformPolicy(lectureId);
  return policy.status == 'student' && policy.sections[section] == true;
}
function platformClaim(lectureId, studentId) {
  let claimId = lectureId + '--' + studentId;
  return get(/databases/$(database)/documents/lectures_hub/config/claims/$(claimId)).data;
}
function platformOwns(lectureId, studentId) {
  let claim = platformClaim(lectureId, studentId);
  return request.auth != null && claim.ownerUid == request.auth.uid;
}

match /lectures_hub/config {
  allow read: if true;
  allow write: if platformInstructor();

  match /claims/{claimId} {
    // Needed for the name selector. These documents contain no plaintext password.
    allow read: if request.auth != null;
    allow create: if request.auth != null
      && request.resource.data.ownerUid == request.auth.uid
      && claimId == request.resource.data.lectureId + '--' + request.resource.data.studentId
      && platformLive(request.resource.data.lectureId, request.resource.data.section)
      && request.resource.data.codeHash == platformPolicy(request.resource.data.lectureId).codeHash;
    allow update: if request.auth != null
      && resource.data.ownerUid == request.auth.uid
      && request.resource.data.ownerUid == resource.data.ownerUid
      && request.resource.data.lectureId == resource.data.lectureId
      && request.resource.data.studentId == resource.data.studentId
      && request.resource.data.section == resource.data.section
      && request.resource.data.name == resource.data.name
      && request.resource.data.diff(resource.data).affectedKeys()
        .hasOnly(['lastSeen', 'sessionId', 'codeHash']);
    allow delete: if platformInstructor();
  }
}
